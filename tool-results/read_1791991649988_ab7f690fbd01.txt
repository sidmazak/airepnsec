    69→function friendlyError(err: unknown, fallback: string): string {
    70→  if (err instanceof ApiClientError && err.message) return err.message

    76→  const n = Number.parseFloat(v)
    77→  return Number.isFinite(n) && n >= 0 ? Math.round(n * 100) : -1

    89→function RecordPaymentDialog({
    90→  invoice, onClose, onChanged,
    91→}: { invoice: Invoice | null; onClose: () => void; onChanged: () => void }) {
    92→  const { toast } = useToast()
    93→  const [amount, setAmount] = useState('')
    94→  const [method, setMethod] = useState('card')
    95→  const [reference, setReference] = useState('')
    96→  const [busy, setBusy] = useState(false)
    97→  const [error, setError] = useState('')
    98→
    99→  useEffect(() => {
   100→    if (invoice) {
   101→      setAmount(((invoice.outstandingCents ?? 0) / 100).toFixed(2))
   102→      setMethod('card')
   103→      setReference('')
   104→      setError('')
   105→    }
   106→  }, [invoice])
   107→
   108→  const submit = async (e: React.FormEvent) => {
   109→    e.preventDefault()
   110→    if (!invoice) return
   111→    const cents = dollarsToCents(amount)
   112→    if (cents < 0) { setError('Enter a valid amount in dollars, e.g. 42.50'); return }
   113→    setBusy(true)
   114→    setError('')
   115→    try {
   116→      await api.post(`/api/invoices/${invoice.id}/payments`, {
   117→        amountCents: cents,
   118→        method,
   119→        reference: reference.trim() || undefined,
   120→      })
   121→      toast({ title: 'Payment recorded', description: `${formatMoney(cents)} verified against ${invoice.number}` })
   122→      onChanged()
   123→      onClose()
   124→    } catch (err) {
   125→      setError(friendlyError(err, 'Could not record payment'))
   126→    } finally {
   127→      setBusy(false)
   128→    }
   129→  }
   130→
   131→  return (
   132→    <Dialog open={!!invoice} onOpenChange={(open) => !open && onClose()}>
   133→      <DialogContent className="sm:max-w-md">
   134→        <DialogHeader>
   135→          <DialogTitle>Record verified payment</DialogTitle>
   136→          <DialogDescription>
   137→            {invoice ? `Money you received yourself and can vouch for. Books against ${invoice.number} immediately.` : ''}
   138→          </DialogDescription>
   139→        </DialogHeader>
   140→        <form onSubmit={submit} className="space-y-3" aria-label="Record payment form">
   141→          {error && <Alert variant="destructive" role="alert"><AlertDescription>{error}</AlertDescription></Alert>}
   142→          <div className="space-y-1.5">
   143→            <Label htmlFor="rp-amount">Amount (USD)</Label>
   144→            <Input id="rp-amount" inputMode="decimal" required value={amount} onChange={(e) => setAmount(e.target.value)} className="min-h-[44px]" />
   145→            {invoice && <p className="text-xs text-muted-foreground">Outstanding: {formatMoney(invoice.outstandingCents ?? 0)}</p>}
   146→          </div>
   147→          <div className="space-y-1.5">
   148→            <Label htmlFor="rp-method">Method</Label>
   149→            <Select value={method} onValueChange={setMethod}>
   150→              <SelectTrigger id="rp-method" className="min-h-[44px] w-full" aria-label="Payment method">
   151→                <SelectValue />
   152→              </SelectTrigger>
   153→              <SelectContent>
   154→                <SelectItem value="cash">Cash</SelectItem>
   155→                <SelectItem value="card">Card</SelectItem>
   156→                <SelectItem value="bank">Bank transfer</SelectItem>
   157→                <SelectItem value="other">Other</SelectItem>
   158→              </SelectContent>
   159→            </Select>
   160→          </div>
   161→          <div className="space-y-1.5">
   162→            <Label htmlFor="rp-reference">Reference (optional)</Label>
   163→            <Input id="rp-reference" value={reference} onChange={(e) => setReference(e.target.value)} placeholder="Receipt #, transfer ID…" className="min-h-[44px]" />
   164→          </div>
   165→          <DialogFooter>
   166→            <Button type="button" variant="outline" className="min-h-[44px]" onClick={onClose}>Cancel</Button>
   167→            <Button type="submit" className="min-h-[44px]" disabled={busy}>
   168→              {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Wallet className="h-4 w-4" aria-hidden />}
   169→              Record payment
   170→            </Button>
   171→          </DialogFooter>
   172→        </form>
   173→      </DialogContent>
   174→    </Dialog>
   175→  )
   176→}

   178→function SimulateWebhookDialog({
   179→  invoice, onClose, onChanged,
   180→}: { invoice: Invoice | null; onClose: () => void; onChanged: () => void }) {
   181→  const { toast } = useToast()
   182→  const [amount, setAmount] = useState('')
   183→  const [busy, setBusy] = useState(false)
   184→  const [error, setError] = useState('')
   185→  const [result, setResult] = useState<string>('')
   186→
   187→  useEffect(() => {
   188→    if (invoice) {
   189→      setAmount(((invoice.outstandingCents ?? 0) / 100).toFixed(2))
   190→      setError('')
   191→      setResult('')
   192→    }
   193→  }, [invoice])
   194→
   195→  const submit = async (e: React.FormEvent) => {
   196→    e.preventDefault()
   197→    if (!invoice) return
   198→    const cents = dollarsToCents(amount)
   199→    if (cents < 0) { setError('Enter a valid amount in dollars, e.g. 42.50'); return }
   200→    setBusy(true)
   201→    setError('')
   202→    setResult('')
   203→    try {
   204→      const res = await api.post<{ result: string }>('/api/payments/simulate-webhook', {
   205→        invoiceNumber: invoice.number,
   206→        amountCents: cents,
   207→      })
   208→      setResult(res.result)
   209→      if (res.result === 'processed') {
   210→        toast({ title: 'Webhook processed', description: `${formatMoney(cents)} applied to ${invoice.number}` })
   211→        onChanged()
   212→      } else {
   213→        toast({ title: `Webhook result: ${res.result}`, description: 'Nothing was booked — see the explanation in the dialog.', variant: res.result === 'rejected' ? 'destructive' : 'default' })
   214→      }
   215→    } catch (err) {
   216→      setError(friendlyError(err, 'Simulation failed'))
   217→    } finally {
   218→      setBusy(false)
   219→    }
   220→  }
   221→
   222→  return (
   223→    <Dialog open={!!invoice} onOpenChange={(open) => !open && onClose()}>
   224→      <DialogContent className="sm:max-w-md">
   225→        <DialogHeader>
   226→          <DialogTitle className="flex items-center gap-2">
   227→            <FlaskConical className="h-4 w-4" aria-hidden /> TEST HARNESS — simulates a signed provider webhook
   228→          </DialogTitle>
   229→          <DialogDescription>
   230→            Sends a real signed webhook event through the exact same processing path a payment provider uses. Not a fake balance: the result is whatever the engine honestly decides.
   231→          </DialogDescription>
   232→        </DialogHeader>
   233→        {invoice && (
   234→          <form onSubmit={submit} className="space-y-3" aria-label="Simulate webhook payment form">
   235→            {error && <Alert variant="destructive" role="alert"><AlertDescription>{error}</AlertDescription></Alert>}
   236→            {result && (
   237→              <Alert role="status">
   238→                <AlertDescription>
   239→                  Result: <strong>{result}</strong>
   240→                  {result === 'processed' && ' — payment applied and ledger posted.'}
   241→                  {result === 'duplicate' && ' — this event was already processed (idempotency). Nothing double-booked.'}
   242→                  {result === 'rejected' && ' — the engine refused it (unknown invoice or invalid amount).'}
   243→                </AlertDescription>
   244→              </Alert>
   245→            )}
   246→            <div className="space-y-1.5">
   247→              <Label htmlFor="sw-amount">Amount (USD) for {invoice.number}</Label>
   248→              <Input id="sw-amount" inputMode="decimal" required value={amount} onChange={(e) => setAmount(e.target.value)} className="min-h-[44px]" />
   249→            </div>
   250→            <p className="text-xs text-muted-foreground">
   251→              Real payment providers POST to <code>/api/payments/webhook</code> with an HMAC signature header (<code>x-signature</code>). This button builds and signs the same payload locally so you can test end-to-end without a provider account.
   252→            </p>
   253→            <DialogFooter>
   254→              <Button type="button" variant="outline" className="min-h-[44px]" onClick={onClose}>Close</Button>
   255→              <Button type="submit" className="min-h-[44px]" disabled={busy}>
   256→                {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <FlaskConical className="h-4 w-4" aria-hidden />}
   257→                Send signed test event
   258→              </Button>
   259→            </DialogFooter>
   260→          </form>
   261→        )}
   262→      </DialogContent>
   263→    </Dialog>
   264→  )
   265→}

   267→function InvoiceDetailDialog({
   268→  invoiceId, onClose, onChanged,
   269→}: { invoiceId: string | null; onClose: () => void; onChanged: () => void }) {
   270→  const session = useAppStore((s) => s.session)
   271→  const { toast } = useToast()
   272→  const isAdmin = session?.user?.role === 'owner' || session?.user?.role === 'admin'
   273→  const [invoice, setInvoice] = useState<Invoice | null>(null)
   274→  const [loading, setLoading] = useState(false)
   275→  const [error, setError] = useState('')
   276→  const [recordOpen, setRecordOpen] = useState(false)
   277→  const [simulateOpen, setSimulateOpen] = useState(false)
   278→  const [voidOpen, setVoidOpen] = useState(false)
   279→  const [busy, setBusy] = useState(false)
   280→
   281→  const load = useCallback(async (id: string) => {
   282→    setLoading(true)
   283→    setError('')
   284→    try {
   285→      const res = await api.get<Invoice>(`/api/invoices/${id}`)
   286→      setInvoice(res)
   287→    } catch (err) {
   288→      setError(friendlyError(err, 'Could not load invoice'))
   289→    } finally {
   290→      setLoading(false)
   291→    }
   292→  }, [])
   293→
   294→  useEffect(() => {
   295→    if (invoiceId) { setInvoice(null); setRecordOpen(false); setSimulateOpen(false); setVoidOpen(false); void load(invoiceId) }
   296→  }, [invoiceId, load])
   297→
   298→  const refreshDetail = () => {
   299→    if (invoiceId) void load(invoiceId)
   300→    onChanged()
   301→  }
   302→
   303→  const doVoid = async () => {
   304→    if (!invoice) return
   305→    setBusy(true)
   306→    try {
   307→      await api.patch(`/api/invoices/${invoice.id}`, { action: 'void' })
   308→      toast({ title: 'Invoice voided', description: `${invoice.number} will not be collected.` })
   309→      setVoidOpen(false)
   310→      refreshDetail()
   311→    } catch (err) {
   312→      toast({ title: 'Could not void invoice', description: friendlyError(err, 'Void failed'), variant: 'destructive' })
   313→    } finally {
   314→      setBusy(false)
   315→    }
   316→  }
   317→
   318→  return (
   319→    <>
   320→      <Dialog open={!!invoiceId} onOpenChange={(open) => !open && onClose()}>
   321→        <DialogContent className="sm:max-w-2xl">
   322→          <DialogHeader>
   323→            <DialogTitle>{invoice ? `Invoice ${invoice.number}` : 'Invoice'}</DialogTitle>
   324→            <DialogDescription>{invoice?.customer?.name ?? ''}</DialogDescription>
   325→          </DialogHeader>
   326→
   327→          {loading && (
   328→            <div className="space-y-2" aria-busy="true">
   329→              <Skeleton className="h-8 w-full" /><Skeleton className="h-24 w-full" /><Skeleton className="h-24 w-full" />
   330→            </div>
   331→          )}
   332→          {error && (
   333→            <div className="space-y-3">
   334→              <Alert variant="destructive" role="alert"><AlertDescription>{error}</AlertDescription></Alert>
   335→              {invoiceId && <Button variant="outline" className="min-h-[44px]" onClick={() => void load(invoiceId)}><RefreshCw className="h-4 w-4" aria-hidden /> Retry</Button>}
   336→            </div>
   337→          )}
   338→
   339→          {invoice && !loading && (
   340→            <div className="space-y-4">
   341→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 text-sm">
   342→                <div><p className="text-xs text-muted-foreground">Status</p><InvoiceStatusBadge status={invoice.status} /></div>
   343→                <div><p className="text-xs text-muted-foreground">Issued</p><p>{formatDate(invoice.issueDate)}</p></div>
   344→                <div><p className="text-xs text-muted-foreground">Due</p><p>{formatDate(invoice.dueDate)}</p></div>
   345→                <div><p className="text-xs text-muted-foreground">Outstanding</p>
   346→                  <p className={cn('font-semibold', (invoice.outstandingCents ?? 0) > 0 ? 'text-amber-700 dark:text-amber-300' : 'text-emerald-700 dark:text-emerald-300')}>
   347→                    {formatMoney(invoice.outstandingCents ?? 0)}
   348→                  </p>
   349→                </div>
   350→              </div>
   351→
   352→              <div>
   353→                <p className="text-sm font-medium mb-1">Line items</p>
   354→                <div className="rounded-md border max-h-40 overflow-y-auto">
   355→                  <Table>
   356→                    <TableHeader><TableRow><TableHead>Description</TableHead><TableHead className="text-right">Qty</TableHead><TableHead className="text-right">Unit</TableHead><TableHead className="text-right">Total</TableHead></TableRow></TableHeader>
   357→                    <TableBody>
   358→                      {(invoice.items ?? []).map((it) => (
   359→                        <TableRow key={it.id}>
   360→                          <TableCell>{it.description}</TableCell>
   361→                          <TableCell className="text-right">{it.qty}</TableCell>
   362→                          <TableCell className="text-right">{formatMoney(it.unitPriceCents)}</TableCell>
   363→                          <TableCell className="text-right">{formatMoney(it.totalCents)}</TableCell>
   364→                        </TableRow>
   365→                      ))}
   366→                    </TableBody>
   367→                  </Table>
   368→                </div>
   369→                <div className="mt-2 text-sm space-y-0.5 text-right">
   370→                  <p className="text-muted-foreground">Subtotal {formatMoney(invoice.subtotalCents)} · Tax {formatMoney(invoice.taxCents)}</p>
   371→                  <p className="font-semibold">Total {formatMoney(invoice.totalCents)} · Paid {formatMoney(invoice.paidCents)}</p>
   372→                </div>
   373→              </div>
   374→
   375→              <div>
   376→                <p className="text-sm font-medium mb-1">Payment history</p>
   377→                {(invoice.payments ?? []).length === 0 ? (
   378→                  <p className="text-sm text-muted-foreground py-2">No payments recorded or claimed yet.</p>
   379→                ) : (
   380→                  <div className="rounded-md border max-h-40 overflow-y-auto">
   381→                    <Table>
   382→                      <TableHeader><TableRow><TableHead>When</TableHead><TableHead>Method</TableHead><TableHead>Reference</TableHead><TableHead>Status</TableHead><TableHead className="text-right">Amount</TableHead></TableRow></TableHeader>
   383→                      <TableBody>
   384→                        {(invoice.payments ?? []).map((p) => (

   386→                            <TableCell className="whitespace-nowrap">{formatDateTime(p.createdAt)}</TableCell>
   387→                            <TableCell>{p.method || '—'}</TableCell>
   388→                            <TableCell className="max-w-[140px] truncate" title={p.reference}>{p.reference || '—'}</TableCell>
   389→                            <TableCell><span className="flex flex-col gap-1"><PaymentStatusBadge status={p.status} /><SourceTag source={p.source} /></span></TableCell>
   390→                            <TableCell className="text-right">{formatMoney(p.amountCents)}</TableCell>
   391→                          </TableRow>
   392→                        ))}
   393→                      </TableBody>
   394→                    </Table>
   395→                  </div>
   396→                )}
   397→              </div>
   398→
   399→              {invoice.notes && (
   400→                <p className="text-sm text-muted-foreground"><span className="font-medium text-foreground">Notes:</span> {invoice.notes}</p>
   401→              )}
   402→
   403→              <Separator />
   404→              <div className="flex flex-wrap gap-2">
   405→                <Button
   406→                  className="min-h-[44px]"
   407→                  disabled={!isAdmin || invoice.status === 'void'}
   408→                  onClick={() => setRecordOpen(true)}
   409→                  title={isAdmin ? undefined : 'Owner or admin only'}
   410→                >
   411→                  <Wallet className="h-4 w-4" aria-hidden /> Record verified payment
   412→                </Button>
   413→                <Button
   414→                  variant="outline"
   415→                  className="min-h-[44px]"
   416→                  disabled={!isAdmin || invoice.status === 'void'}
   417→                  onClick={() => setSimulateOpen(true)}
   418→                  title={isAdmin ? 'Test harness' : 'Owner or admin only'}
   419→                >
   420→                  <FlaskConical className="h-4 w-4" aria-hidden /> Simulate webhook payment
   421→                </Button>
   422→                <Button
   423→                  variant="outline"
   424→                  className="min-h-[44px] text-destructive hover:text-destructive"
   425→                  disabled={!isAdmin || invoice.status === 'void'}
   426→                  onClick={() => setVoidOpen(true)}
   427→                  title={isAdmin ? undefined : 'Owner or admin only'}
   428→                >
   429→                  <Ban className="h-4 w-4" aria-hidden /> Void invoice
   430→                </Button>
   431→                {!isAdmin && (
   432→                  <p className="text-xs text-muted-foreground w-full">Payment recording, webhook simulation and voiding are restricted to owner/admin accounts.</p>

   434→              </div>
   435→            </div>
   436→          )}
   437→        </DialogContent>
   438→      </Dialog>
   439→
   440→      {invoice && (
   441→        <>
   442→          <RecordPaymentDialog invoice={recordOpen ? invoice : null} onClose={() => setRecordOpen(false)} onChanged={refreshDetail} />

   444→          <AlertDialog open={voidOpen} onOpenChange={(open) => !open && setVoidOpen(false)}>
   445→            <AlertDialogContent>
   446→              <AlertDialogHeader>
   447→                <AlertDialogTitle>Void invoice {invoice.number}?</AlertDialogTitle>
   448→                <AlertDialogDescription>
   449→                  A voided invoice is permanently un-collectible. Existing ledger charge entries are reversed automatically. This cannot be undone.
   450→                </AlertDialogDescription>
   451→              </AlertDialogHeader>
   452→              <AlertDialogFooter>
   453→                <AlertDialogCancel className="min-h-[44px]">Keep invoice</AlertDialogCancel>
   454→                <AlertDialogAction
   455→                  className="min-h-[44px] bg-destructive text-white hover:bg-destructive/90"
   456→                  disabled={busy}
   457→                  onClick={(e) => { e.preventDefault(); void doVoid() }}
   458→                >
   459→                  {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Trash2 className="h-4 w-4" aria-hidden />}
   460→                  Void it
   461→                </AlertDialogAction>
   462→              </AlertDialogFooter>
   463→            </AlertDialogContent>
   464→          </AlertDialog>
   465→        </>
   466→      )}
   467→    </>
   468→  )
   469→}

   473→function VerifyQueueTab({ onChanged }: { onChanged: () => void }) {
   474→  const session = useAppStore((s) => s.session)
