    56→    56→
    57→    57→export function AiSection() {
    58→    58→  const { toast } = useToast()
    59→    59→  const session = useAppStore((s) => s.session)
    60→    60→  const canManage = session?.user?.role === 'owner' || session?.user?.role === 'admin'
    61→    61→
    62→    62→  const [loading, setLoading] = useState(true)
    63→    63→  const [error, setError] = useState('')
    64→    64→  const [capabilities, setCapabilities] = useState<Record<string, ProviderConfigPublic>>({})
    65→    65→  const [capabilityInfo, setCapabilityInfo] = useState<Record<string, CapabilityInfo>>({})
    66→    66→  const [dataMode, setDataMode] = useState('hybrid')
    67→    67→
    68→    68→  const [drafts, setDrafts] = useState<Record<string, Draft>>({})
    69→    69→  const [savingCap, setSavingCap] = useState<string | null>(null)
    70→    70→  const [testingCap, setTestingCap] = useState<string | null>(null)
    71→    71→  const [testResults, setTestResults] = useState<Record<string, TestResult>>({})
    72→    72→
    73→    73→  const [tab, setTab] = useState('providers')
    74→    74→  const [logs, setLogs] = useState<AiLog[]>([])
    75→    75→  const [logsLoading, setLogsLoading] = useState(false)
    76→    76→  const [logsError, setLogsError] = useState('')
    77→    77→
    78→    78→  const [channels, setChannels] = useState<{ email: boolean; sms: boolean } | null>(null)
    79→    79→  const [channelsBusy, setChannelsBusy] = useState(false)
    80→    80→
    81→    81→  const load = useCallback(async () => {
    82→    82→    setLoading(true)
    83→    83→    try {
    84→    84→      const d = await api.get<{ capabilities: Record<string, ProviderConfigPublic>; dataMode: string; capabilityInfo: Record<string, CapabilityInfo> }>('/api/providers')
    85→    85→      setCapabilities(d.capabilities)
    86→    86→      setCapabilityInfo(d.capabilityInfo ?? {})
    87→    87→      setDataMode(d.dataMode)
    88→    88→      setDrafts(Object.fromEntries(
    89→    89→        Object.entries(d.capabilities).map(([cap, cfg]) => [cap, {
    90→    90→          provider: cfg.provider, baseUrl: cfg.baseUrl, model: cfg.model, apiKey: null, enabled: cfg.enabled,
    91→    91→        }]),
    92→    92→      ))
    93→    93→      setError('')
    94→    94→    } catch (err) {
    95→    95→      setError(err instanceof Error ? err.message : 'Could not load AI provider settings')
    96→    96→    } finally {
    97→    97→      setLoading(false)
    98→    98→    }
    99→    99→  }, [])
   100→   100→
   101→   101→  const loadLogs = useCallback(async () => {
   102→   102→    setLogsLoading(true)
   103→   103→    try {
   104→   104→      const d = await api.get<{ logs: AiLog[] }>('/api/ai/logs?limit=30')
   105→   105→      setLogs(d.logs)
   106→   106→      setLogsError('')
   107→   107→    } catch (err) {
   108→   108→      setLogsError(err instanceof Error ? err.message : 'Could not load the activity log')
   109→   109→    } finally {
   110→   110→      setLogsLoading(false)
   111→   111→    }
   112→   112→  }, [])
   113→   113→
   114→   114→  const loadChannels = useCallback(async () => {
   115→   115→    try {
   116→   116→      const d = await api.get<{ email: boolean; sms: boolean }>('/api/settings/channels')
   117→   117→      setChannels(d)
   118→   118→    } catch { /* non-fatal */ }
   119→   119→  }, [])
   120→   120→
   121→   121→  useEffect(() => { void load() }, [load])
   122→   122→  useEffect(() => { void loadChannels() }, [loadChannels])
   123→   123→  useEffect(() => { if (tab === 'logs') void loadLogs() }, [tab, loadLogs])
   124→   124→
   125→   125→  const setDataModeValue = async (mode: string) => {
   126→   126→    if (!canManage) return
   127→   127→    try {
   128→   128→      await api.put('/api/settings/data-mode', { dataMode: mode })
   129→   129→      setDataMode(mode)
   130→   130→      toast({
   131→   131→        title: 'Data mode updated',
   132→   132→        description: mode === 'local_only' ? 'Cloud AI calls are now blocked server-side.' : 'Cloud AI calls follow your per-capability settings.',
   133→   133→      })
   134→   134→    } catch (err) {
   135→   135→      toast({ title: 'Could not change data mode', description: err instanceof Error ? err.message : 'Unknown error', variant: 'destructive' })
   136→   136→    }
   137→   137→  }
   138→   138→
   139→   139→  const setDraft = (cap: string, patch: Partial<Draft>) => {
   140→   140→    setDrafts((d) => ({ ...d, [cap]: { ...d[cap], ...patch } }))
   141→   141→  }
   142→   142→
   143→   143→  const saveCapability = async (cap: string) => {
   144→   144→    const draft = drafts[cap]
   145→   145→    if (!draft) return
   146→   146→    setSavingCap(cap)
   147→   147→    try {
   148→   148→      const body: Record<string, unknown> = {
   149→   149→        provider: draft.provider,
   150→   150→        baseUrl: draft.baseUrl.trim(),
   151→   151→        model: draft.model.trim(),
   152→   152→        enabled: draft.enabled,
   153→   153→      }
   154→   154→      if (draft.apiKey !== null) body.apiKey = draft.apiKey // '' clears, non-empty replaces, null keeps
   155→   155→      const d = await api.put<{ config: ProviderConfigPublic }>(`/api/providers/${cap}`, body)
   156→   156→      setCapabilities((c) => ({ ...c, [cap]: d.config }))
   157→   157→      setDrafts((dr) => ({ ...dr, [cap]: { ...dr[cap], provider: d.config.provider, baseUrl: d.config.baseUrl, model: d.config.model, apiKey: null, enabled: d.config.enabled } }))
   158→   158→      toast({ title: `${capabilityInfo[cap]?.label ?? cap} saved` })
   159→   159→    } catch (err) {
   160→   160→      toast({ title: 'Save failed', description: err instanceof Error ? err.message : 'Unknown error', variant: 'destructive' })
   161→   161→    } finally {
   162→   162→      setSavingCap(null)
   163→   163→    }
   164→   164→  }
   165→   165→
   166→   166→  const testCapability = async (cap: string) => {
   167→   167→    setTestingCap(cap)
   168→   168→    try {
   169→   169→      const r = await api.post<TestResult>(`/api/providers/${cap}/test`)
   170→   170→      setTestResults((t) => ({ ...t, [cap]: r }))
   171→   171→      await load()
   172→   172→    } catch (err) {
   173→   173→      setTestResults((t) => ({
   174→   174→        ...t,
   175→   175→        [cap]: { ok: false, message: err instanceof Error ? err.message : 'Test failed', latencyMs: 0 },
   176→   176→      }))
   177→   177→    } finally {
   178→   178→      setTestingCap(null)
   179→   179→    }
   180→   180→  }
   181→   181→
   182→   182→  const toggleChannel = async (key: 'email' | 'sms', value: boolean) => {
   183→   183→    if (!channels || !canManage) return
   184→   184→    setChannelsBusy(true)
   185→   185→    const next = { ...channels, [key]: value }
   186→   186→    try {
   187→   187→      const d = await api.put<{ email: boolean; sms: boolean }>('/api/settings/channels', next)
   188→   188→      setChannels(d)
   189→   189→      toast({
   190→   190→        title: value ? `${key.toUpperCase()} channel marked configured` : `${key.toUpperCase()} channel disabled`,
   191→   191→        description: value ? 'Queued outbox messages will use it once a sender is wired.' : undefined,
   192→   192→      })
   193→   193→    } catch (err) {
   194→   194→      toast({ title: 'Could not update channels', description: err instanceof Error ? err.message : 'Unknown error', variant: 'destructive' })
   195→   195→    } finally {
   196→   196→      setChannelsBusy(false)
   197→   197→    }
   198→   198→  }
   199→   199→
   200→   200→  if (loading) {
   201→   201→    return (
   202→   202→      <div className="space-y-4" aria-label="Loading AI settings">
   203→   203→        <Skeleton className="h-8 w-64" />
   204→   204→        <div className="grid gap-4 lg:grid-cols-3">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-32 w-full" />)}</div>
   205→   205→        <div className="grid gap-4 lg:grid-cols-2">{[0, 1, 2, 3].map((i) => <Skeleton key={i} className="h-64 w-full" />)}</div>
   206→   206→      </div>
   207→   207→    )
   208→   208→  }
   209→   209→
   210→   210→  if (error) {
   211→   211→    return (
   212→   212→      <Alert variant="destructive" role="alert">
   213→   213→        <AlertDescription className="space-y-2">
   214→   214→          <p>{error}</p>
   215→   215→          <Button size="sm" variant="outline" onClick={() => void load()} className="min-h-[44px]">Retry</Button>
   216→   216→        </AlertDescription>
   217→   217→      </Alert>
   218→   218→    )
   219→   219→  }
   220→   220→
   221→   221→  const caps = Object.entries(capabilities)
   222→   222→
   223→   223→  return (
   224→   224→    <div className="space-y-4" aria-label="AI providers and privacy">
   225→   225→      <div className="flex flex-wrap items-center gap-2">
   226→   226→        <div className="mr-auto">
   227→   227→          <h2 className="text-lg font-semibold tracking-tight">AI Providers &amp; privacy</h2>
   228→   228→          <p className="text-sm text-muted-foreground">Which models run each capability, where your data flows, and proof that it works.</p>
   229→   229→        </div>
   230→   230→        {!canManage && <Badge variant="outline">read-only — owner/admin can edit</Badge>}
   231→   231→      </div>
   232→   232→
   233→   233→      <Tabs value={tab} onValueChange={setTab}>
   234→   234→        <TabsList aria-label="AI settings tabs">
   235→   235→          <TabsTrigger value="providers" className="min-h-[44px]"><Bot className="h-4 w-4 mr-1.5" aria-hidden /> Providers</TabsTrigger>
   236→   236→          <TabsTrigger value="logs" className="min-h-[44px]"><RefreshCw className="h-4 w-4 mr-1.5" aria-hidden /> AI activity log</TabsTrigger>
   237→   237→        </TabsList>
   238→   238→
   239→   239→        <TabsContent value="providers" className="space-y-4 mt-4">
   240→   240→          {/* data mode */}
   241→   241→          <Card>
   242→   242→            <CardHeader>
   243→   243→              <CardTitle className="text-base flex items-center gap-2"><Globe className="h-4 w-4" aria-hidden /> Data mode</CardTitle>
   244→   244→              <CardDescription>Controls whether any cloud AI may be contacted, org-wide.</CardDescription>
   245→   245→            </CardHeader>
   246→   246→            <CardContent>
   247→   247→              <div className="grid gap-3 sm:grid-cols-3" role="radiogroup" aria-label="Data mode">
   248→   248→                {DATA_MODES.map((m) => {
   249→   249→                  const selected = dataMode === m.id
   250→   250→                  return (
   251→   251→                    <button
   252→   252→                      key={m.id}
   253→   253→                      type="button"
   254→   254→                      role="radio"
   255→   255→                      aria-checked={selected}
   256→   256→                      disabled={!canManage}
   257→   257→                      onClick={() => void setDataModeValue(m.id)}
   258→   258→                      className={cn(
   259→   259→                        'text-left rounded-lg border p-3 min-h-[44px] transition-colors disabled:opacity-60',
   260→   260→                        selected ? 'border-primary bg-muted/50' : 'hover:bg-muted/40',
   261→   261→                      )}
   262→   262→                    >
   263→   263→                      <div className="flex items-center gap-2">
   264→   264→                        {m.id === 'local_only'
   265→   265→                          ? <ShieldCheck className={cn('h-4 w-4', selected ? 'text-emerald-700' : 'text-muted-foreground')} aria-hidden />
   266→   266→                          : <ShieldAlert className={cn('h-4 w-4', selected && m.id !== 'hybrid' ? 'text-amber-700' : 'text-muted-foreground')} aria-hidden />}
   267→   267→                        <span className="font-medium text-sm">{m.title}</span>
   268→   268→                      </div>
   269→   269→                      <p className="text-xs text-muted-foreground mt-1">{m.description}</p>
   270→   270→                    </button>
   271→   271→                  )
   272→   272→                })}
   273→   273→              </div>
   274→   274→              {dataMode === 'local_only' && (
   275→   275→                <p className="mt-2 text-xs text-emerald-800 flex items-center gap-1.5">
   276→   276→                  <ShieldCheck className="h-3.5 w-3.5" aria-hidden /> Strict mode is enforced server-side — cloud calls are blocked even if configured.
   277→   277→                </p>
   278→   278→              )}
   279→   279→            </CardContent>
   280→   280→          </Card>
   281→   281→
   282→   282→          {/* capability cards */}
   283→   283→          <div className="grid gap-4 lg:grid-cols-2">
   284→   284→            {caps.map(([cap, cfg]) => {
   285→   285→              const info = capabilityInfo[cap]
   286→   286→              const draft = drafts[cap] ?? { provider: cfg.provider, baseUrl: cfg.baseUrl, model: cfg.model, apiKey: null, enabled: cfg.enabled }
   287→   287→              const test = testResults[cap]
   288→   288→              const cloud = isCloudProvider(draft.provider)
   289→   289→              const needsBaseUrl = draft.provider === 'ollama' || draft.provider === 'openai_compatible'
   290→   290→              const needsKey = draft.provider === 'openai_compatible'
   291→   291→              return (
   292→   292→                <Card key={cap}>
   293→   293→                  <CardHeader>
   294→   294→                    <div className="flex items-start gap-2">
   295→   295→                      <div className="min-w-0 flex-1">
   296→   296→                        <CardTitle className="text-base">{info?.label ?? cap}</CardTitle>
   297→   297→                        <CardDescription>{info?.description}</CardDescription>
   298→   298→                      </div>
   299→   299→                      <label className="flex items-center gap-2 text-sm shrink-0">
   300→   300→                        <Switch
   301→   301→                          checked={draft.enabled}
   302→   302→                          disabled={!canManage}
   303→   303→                          onCheckedChange={(v) => setDraft(cap, { enabled: v })}
   304→   304→                          aria-label={`Enable ${info?.label ?? cap}`}
   305→   305→                        />
   306→   306→                        {draft.enabled ? 'On' : 'Off'}
   307→   307→                      </label>
   308→   308→                    </div>
   309→   309→                  </CardHeader>
   310→   310→                  <CardContent className="space-y-3">
   311→   311→                    <div className="space-y-1.5">
   312→   312→                      <Label htmlFor={`prov-${cap}`}>Provider</Label>
   313→   313→                      <Select value={draft.provider} disabled={!canManage} onValueChange={(v) => setDraft(cap, { provider: v })}>
   314→   314→                        <SelectTrigger id={`prov-${cap}`} className="min-h-[44px]" aria-label={`Provider for ${info?.label ?? cap}`}>
   315→   315→                          <SelectValue />
   316→   316→                        </SelectTrigger>
   317→   317→                        <SelectContent>
   318→   318→                          {(info?.kinds ?? Object.keys(PROVIDER_LABELS)).map((k) => (
   319→   319→                            <SelectItem key={k} value={k}>{PROVIDER_LABELS[k] ?? k}</SelectItem>
   320→   320→                          ))}
   321→   321→                        </SelectContent>
   322→   322→                      </Select>
   323→   323→                    </div>
   324→   324→
   325→   325→                    {needsBaseUrl && (
   326→   326→                      <div className="space-y-1.5">
   327→   327→                        <Label htmlFor={`url-${cap}`}>Base URL</Label>
   328→   328→                        <Input
   329→   329→                          id={`url-${cap}`} value={draft.baseUrl} disabled={!canManage}
   330→   330→                          onChange={(e) => setDraft(cap, { baseUrl: e.target.value })}
   331→   331→                          placeholder="http://localhost:11434"
   332→   332→                          className="min-h-[44px]"
   333→   333→                        />
   334→   334→                      </div>
   335→   335→                    )}
   336→   336→
   337→   337→                    {(needsBaseUrl || draft.provider === 'managed') && (
   338→   338→                      <div className="space-y-1.5">
   339→   339→                        <Label htmlFor={`model-${cap}`}>Model</Label>
   340→   340→                        <Input
   341→   341→                          id={`model-${cap}`} value={draft.model} disabled={!canManage}
   342→   342→                          onChange={(e) => setDraft(cap, { model: e.target.value })}
   343→   343→                          placeholder={draft.provider === 'managed' ? '(gateway default)' : cap === 'embedding' ? 'e.g. nomic-embed-text / text-embedding-3-small' : 'e.g. llama3.1:8b'}
   344→   344→                          className="min-h-[44px]"
   345→   345→                        />
   346→   346→                      </div>
   347→   347→                    )}
   348→   348→
   349→   349→                    {needsKey && (
   350→   350→                      <div className="space-y-1.5">
   351→   351→                        <Label htmlFor={`key-${cap}`} className="flex items-center gap-1.5"><KeyRound className="h-3.5 w-3.5" aria-hidden /> API key</Label>
   352→   352→                        <Input
   353→   353→                          id={`key-${cap}`} type="password" disabled={!canManage}
   354→   354→                          value={draft.apiKey ?? ''}
   355→   355→                          onChange={(e) => setDraft(cap, { apiKey: e.target.value })}
   356→   356→                          placeholder={cfg.apiKeyPresent ? '•••••••• (saved — leave blank to keep)' : 'Paste key'}
   357→   357→                          autoComplete="off"
   358→   358→                          className="min-h-[44px]"
   359→   359→                        />
   360→   360→                        <p className="text-xs text-muted-foreground">
   361→   361→                          {cfg.apiKeyPresent
   362→   362→                            ? 'A key is saved. Leave blank to keep it — type a new one to replace it, or empty the field and save to remove it.'
   363→   363→                            : 'Stored server-side, never echoed back to the browser.'}
   364→   364→                        </p>
   365→   365→                      </div>
   366→   366→                    )}
   367→   367→
   368→   368→                    <div className="rounded-md border p-2.5 text-xs flex items-start gap-2" aria-label="Data flow">
   369→   369→                      {cloud
   370→   370→                        ? <ShieldAlert className="h-4 w-4 text-amber-600 shrink-0 mt-0.5" aria-hidden />
   371→   371→                        : <ShieldCheck className="h-4 w-4 text-emerald-600 shrink-0 mt-0.5" aria-hidden />}
   372→   372→                      <span className={cloud ? 'text-amber-900' : 'text-emerald-900'}>{cfg.processesDataWhere || `Data is processed ${cloud ? 'by the external provider' : 'locally'}.`}</span>
   373→   373→                    </div>
   374→   374→
   375→   375→                    <div className="flex flex-wrap items-center gap-2">
   376→   376→                      <Button
   377→   377→                        size="sm" onClick={() => void saveCapability(cap)} disabled={!canManage || savingCap === cap}
   378→   378→                        className="min-h-[44px]" aria-label={`Save ${info?.label ?? cap} configuration`}
   379→   379→                      >
   380→   380→                        {savingCap === cap ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <PlugZap className="h-4 w-4" aria-hidden />}
   381→   381→                        Save
   382→   382→                      </Button>
   383→   383→                      <Button
   384→   384→                        size="sm" variant="outline" onClick={() => void testCapability(cap)}
   385→   385→                        disabled={testingCap === cap || !canManage}
   386→   386→                        className="min-h-[44px]" aria-label={`Test connection for ${info?.label ?? cap}`}
   387→   387→                      >
   388→   388→                        {testingCap === cap ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <RefreshCw className="h-4 w-4" aria-hidden />}
   389→   389→                        Test connection
   390→   390→                      </Button>
   391→   391→                    </div>
   392→   392→
   393→   393→                    {test && (
